Supabase SQL Schema (1-Click Setup)

Paste this SQL into your Supabase SQL Editor to automatically create tables with read/write access.

-- 1. Create Projects Table
create table if not exists projects (
  id text primary key,
  title text not null,
  category text,
  tag text,
  image text,
  client text,
  timeline text,
  metrics text,
  description text,
  deliverables jsonb,
  created_at timestamp default now()
);

-- 2. Create Testimonials Table
create table if not exists testimonials (
  id text primary key,
  name text not null,
  role text,
  initials text,
  rating int default 5,
  quote text,
  created_at timestamp default now()
);

-- 3. Create Inquiries Table
create table if not exists inquiries (
  id text primary key,
  name text not null,
  email text,
  phone text,
  service text,
  message text,
  status text default 'new',
  date text,
  created_at timestamp default now()
);

-- Enable Public Read & Insert Access for Website Visitors
alter table projects enable row level security;
alter table testimonials enable row level security;
alter table inquiries enable row level security;

create policy "Public read projects" on projects for select using (true);
create policy "Public modify projects" on projects for all using (true);

create policy "Public read testimonials" on testimonials for select using (true);
create policy "Public modify testimonials" on testimonials for all using (true);

create policy "Public read inquiries" on inquiries for select using (true);
create policy "Public modify inquiries" on inquiries for all using (true);
